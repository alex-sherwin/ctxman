```ts
// Resolves the absolute data directory. Pure node:path — safe for content.config.ts, scripts,
// Vitest, and SSR runtime. Relative values resolve against process.cwd() (npm scripts run at repo root).
import path from 'node:path';
import { pathToFileURL } from 'node:url';
import type { DatasetId } from '../datasets/registry.ts';

export const DATA_DIR_ENV = 'CTXMAN_DATA_DIR';

export function resolveDataDir(env: NodeJS.ProcessEnv = process.env): string {
  const raw = env[DATA_DIR_ENV];
  return path.resolve(process.cwd(), raw !== undefined && raw !== '' ? raw : 'data');
}

/** Absolute path of data/<dataset>. */
export function datasetDir(dataset: DatasetId, dataDir: string = resolveDataDir()): string {
  return path.join(dataDir, dataset);
}

/**
 * Directory as a file: URL with trailing slash. Pass THIS (not a string) to the astro glob loader
 * `base` option: a Windows path string like C:\x would be parsed as a URL scheme by `new URL()`.
 */
export function dirUrl(absDir: string): URL {
  const url = pathToFileURL(absDir);
  if (!url.pathname.endsWith('/')) url.pathname += '/';
  return url;
}
```

```ts
// YAML helpers shared by content.config.ts, scripts, and server libs.
// parseYaml mirrors astro's own parser (@astrojs/internal-helpers/yaml) so a file validates
// identically in `astro sync` and in scripts/check-*.ts: unquoted dates become Date objects.
import { parseDocument, stringify } from 'yaml';

export function parseYaml(source: string): unknown {
  const doc = parseDocument(source, { customTags: ['timestamp'], merge: true, schema: 'core' });
  const first = doc.errors[0];
  if (first) throw first;
  return doc.contents === null ? undefined : doc.toJS();
}

/** Deterministic output for files ctxman writes (context yaml): literal block scalars, no line folding. */
export function stringifyYaml(value: unknown): string {
  return stringify(value, { blockQuote: 'literal', lineWidth: 0 });
}
```

```ts
import { existsSync } from 'node:fs';
import { readdir, readFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { defineCollection } from 'astro:content';
import { glob } from 'astro/loaders';
import type { Loader } from 'astro/loaders';
import { OKF_DATASET_IDS } from './datasets/registry.ts';
import type { OkfDatasetId } from './datasets/registry.ts';
import {
  bugSchema,
  contextSchema,
  docSchema,
  featureSchema,
  ideaSchema,
  kbSchema,
  projectSchema,
  rcaSchema,
  referenceSchema,
  skillSchema,
  taskSchema,
} from './datasets/schemas.ts';
import { datasetDir, dirUrl, resolveDataDir } from './lib/data-dir.ts';
import { parseYaml } from './lib/yaml.ts';

const dataDir = resolveDataDir();

/** id = first path segment (the bundle / skill directory name). `entry` is posix, relative to base. */
const dirNameId = ({ entry }: { entry: string }): string => entry.split('/')[0] ?? entry;
/** id = file stem of a top-level yaml file. */
const stemId = ({ entry }: { entry: string }): string => entry.replace(/\.ya?ml$/, '');

/**
 * context/*.yaml loader WITHOUT a file watcher. The forge writes these files at runtime; a glob
 * loader would make `astro dev` push a full browser reload ~0.7s after every save. Runtime reads of
 * contexts go through src/lib/context-io.ts (fs); this collection exists for schema validation in
 * `astro sync` / `astro build` / typecheck. NEVER call getCollection('context') in pages/APIs.
 */
function contextYamlLoader(dir: string): Loader {
  return {
    name: 'ctxman-context-yaml',
    load: async ({ store, parseData, generateDigest, config }) => {
      store.clear();
      const files = existsSync(dir) ? (await readdir(dir)).filter((f) => f.endsWith('.yaml')) : [];
      for (const file of files.sort()) {
        const id = file.slice(0, -'.yaml'.length);
        const filePath = path.join(dir, file);
        const raw = await readFile(filePath, 'utf8');
        const parsed = parseYaml(raw);
        const input =
          parsed !== null && typeof parsed === 'object' ? (parsed as Record<string, unknown>) : {};
        const data = await parseData({ id, data: input, filePath });
        // store.set requires a posix path relative to the site root (like the glob loader stores).
        const rel = path.relative(fileURLToPath(config.root), filePath).split(path.sep).join('/');
        store.set({ id, data, filePath: rel, digest: generateDigest(raw) });
      }
    },
  };
}

const okfBundle = (dataset: OkfDatasetId) =>
```

## E. `src/lib` contracts
Conventions: every module is erasable TS with `.ts` import extensions. "PURE" = no fs, no astro:* (safe in browser islands, Node scripts, Vitest). "NODE" = node:fs/path allowed, no astro:* (safe in scripts + Vitest + SSR). "SERVER-ASTRO" = imports `astro:content` (pages/API only; never imported by scripts, tests, or islands). All paths returned to callers are **posix, relative to the data dir** unless named `abs*`. All functions taking `dataDir` default it to `resolveDataDir()`.

### E.1 `data-dir.ts` [done, NODE]
`DATA_DIR_ENV = 'CTXMAN_DATA_DIR'`; `resolveDataDir(env = process.env): string` (absolute; relative values resolve against `process.cwd()`, default `./data`); `datasetDir(dataset, dataDir?)`; `dirUrl(absDir): URL` (file URL with trailing slash, for glob `base`).

### E.2 `yaml.ts` [done, NODE-safe/PURE]
`parseYaml(source): unknown` — identical options to astro (`customTags: ['timestamp'], merge: true, schema: 'core'`), throws the first `YAMLParseError`. `stringifyYaml(value): string` — `stringify(value, { blockQuote: 'literal', lineWidth: 0 })` (multi-line strings become `|` blocks; VERIFIED).

### E.3 `markdown.ts` [done, SERVER] / `markdown-config.ts` [done, PURE]
`renderMarkdown(md): Promise<string>` — Sätteri + `SHIKI_CONFIG`; `''` for blank input; renderer created once per process. Used for context notes (`ContextLinks.astro`). `SHIKI_CONFIG = { themes: { light: 'github-light', dark: 'github-dark' }, wrap: false } as const`.

- `root-index`: `okf/missing-index` (warning) absent; `okf/index-frontmatter` (error) frontmatter has keys other than `okf_version` or `okf_version !== "0.2"` (string).
- nested `index.md` / any `log.md`: `okf/reserved-frontmatter` (error) has frontmatter.
- `log.md`: `okf/log-heading` (error) a `## ` heading that is not `YYYY-MM-DD`; `okf/log-order` (warning) dates not strictly newest-first.
- `stale_after`: `okf/stale-after-format` (error) raw value not `YYYY-MM-DD` (unquoted YAML date parses to Date at 00:00Z — accept; string must match `/^\d{4}-\d{2}-\d{2}$/`).
- footnotes: `okf/footnote-source` (warning) a footnote reference `[^id]` in the body whose `id` has neither a `[^id]:` definition nor a `sources[].id`.
